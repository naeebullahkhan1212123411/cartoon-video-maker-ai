import express from "express"; import dotenv from "dotenv"; import { fal } from "@fal-ai/client"; import path from "node:path"; import { fileURLToPath } from "node:url"; dotenv.config(); const app = express(); const PORT = process.env.PORT || 3000; const filename = fileURLToPath(import.meta.url); const dirname = path.dirname(filename); app.use(express.json({ limit: "1mb" })); app.use(express.static(path.join(dirname, "public"))); app.get("/api/health", (_req, res) => { res.json({ ok: true, configured: Boolean(process.env.FAL_KEY) }); }); app.post("/api/generate", async (req, res) => { try { if (!process.env.FAL_KEY) { return res.status(503).json({ error: "Server par FAL_KEY API key set karein." }); } const { prompt, style = "Anime", duration = 5, aspectRatio = "16:9" } = req.body ?? {}; if (typeof prompt !== "string" || prompt.trim().length < 5) { return res.status(400).json({ error: "Kam az kam 5 characters ki story likhein." }); } if (prompt.length > 4000) { return res.status(400).json({ error: "Story 4000 characters se chhoti rakhein." }); } const seconds = Number(duration); if (![5, 10, 15].includes(seconds)) { return res.status(400).json({ error: "Duration 5, 10 ya 15 seconds select karein." }); } if (!["16:9", "9:16", "1:1"].includes(aspectRatio)) { return res.status(400).json({ error: "Video shape supported nahi hai." }); } const fullPrompt = `${prompt.trim()}. Visual style: ${String(style).slice(0, 40)} cartoon animation. Family-friendly, colorful background, smooth motion and coherent scenes.`; const result = await fal.subscribe( "fal-ai/wan/v2.7/text-to-video", { input: { prompt: fullPrompt, duration: seconds, aspect_ratio: aspectRatio, resolution: "720p", enable_prompt_expansion: true, enable_safety_checker: true }, logs: true, onQueueUpdate(update) { if (update.status) { console.log("Video status:", update.status); } } } ); const videoUrl = result?.data?.video?.url; if (!videoUrl) { return res.status(502).json({ error: "AI service ne video return nahi ki." }); } res.json({ videoUrl, duration: seconds, style }); } catch (error) { console.error("Generation error:", error?.message || error); res.status(500).json({ error: "Video nahi ban saki. API key, account aur credits check karein." }); } }); app.listen(PORT, () => { console.log(`ToonSpark running on port ${PORT}`); });
